      if (blocked(bx, by, bz, 15)) { console.log(`[overgrowth-mids] ${ds.key} forecourt blocked at every outward nudge — site skipped this build`); continue; }
      const hsteps = [];
      for (let k = 1; k <= 3; k++) {                                   // 3 hop-steps from the rim pad down to the plaza

        db2.material.fog = false; db2.userData.noCull = true;

### OVERGROWTH/ASCENT/SKYFALL_SHAPE walkers + comment (from pre-cut index.html)
// LEVELS 2-4's macro-shape generators, passed as buildNetwork's `shape` param (see buildNetwork, ~index.html:5386).
// Each `place(j, M, slot, rand)` maps a want-list index (0..M-1) within one universe to {ang, dist, zoneIdx} -
// genuinely different spatial patterns from the default's uniform radial-band scatter, while reusing buildNetwork's
// own hub-building/collision-avoidance/connect() machinery verbatim (only the WHERE changes, not the HOW).
const OVERGROWTH_SHAPE = {        // branching tree: BOUGHS boughs per universe, hubs climb outward rung by rung,
  zones: OVERGROWTH_ZONES,        // alternating boughs drift opposite ways so they read as separate limbs, not one ring.
  place(j, M, slot, rand) {
    const BOUGHS = 5, rungsTotal = Math.ceil(M / BOUGHS);
    const b = j % BOUGHS, rung = Math.floor(j / BOUGHS);
    const baseAngle = slot * 0.73 + b * (6.28318 / BOUGHS);
    const ang = baseAngle + (b % 2 ? 1 : -1) * rung * 0.13 + (rand() - 0.5) * 0.2;
    const dist = 72 + rung * 54 + rand() * 20;
    const zoneIdx = rung < rungsTotal / 3 ? 0 : rung < rungsTotal * 2 / 3 ? 1 : 2;
    return { ang, dist, zoneIdx };
  }
};
const ASCENT_SHAPE = {        // single switchback trail: a zigzagging path per universe, hairpinning left/right every
  zones: ASCENT_ZONES,        // leg while still net-drifting forward so it winds most of the way around the massif.
  place(j, M, slot, rand) {
    const leg = Math.floor(j / 2), legDir = (leg % 2 === 0) ? 1 : -1;

  }
